Backend Architecture

Supabase Production Database Schema (PostgreSQL)

Copy and paste this into the Supabase SQL Editor to spin up your cloud backend in 60 seconds.

-- 1. CONTRACTOR PROFILES TABLE
create table if not exists public.contractors (
  id uuid primary key default gen_random_uuid(),
  created_at timestamp with time zone default now(),
  email text unique not null,
  company_name text not null,
  phone text,
  license_number text,
  default_margin_pct numeric default 25.0,
  default_tax_pct numeric default 7.5,
  stripe_customer_id text,
  subscription_status text default 'trial'
);

-- 2. CLIENT RECORDS TABLE
create table if not exists public.customers (
  id uuid primary key default gen_random_uuid(),
  contractor_id uuid references public.contractors(id) on delete cascade,
  created_at timestamp with time zone default now(),
  full_name text not null,
  phone text not null,
  job_address text not null
);

-- 3. QUOTES / ESTIMATES TABLE
create table if not exists public.quotes (
  id uuid primary key default gen_random_uuid(),
  quote_number text unique not null,
  contractor_id uuid references public.contractors(id) on delete cascade,
  customer_id uuid references public.customers(id) on delete set null,
  created_at timestamp with time zone default now(),
  status text default 'draft', -- draft, sent, signed, deposit_paid
  category text,
  subtotal numeric not null default 0.00,
  markup_pct numeric default 25.0,
  markup_amount numeric default 0.00,
  tax_pct numeric default 7.5,
  tax_amount numeric default 0.00,
  total_amount numeric not null default 0.00,
  required_deposit numeric not null default 0.00,
  signature_svg text,
  signed_at timestamp with time zone,
  signer_name text
);

-- 4. ITEM LINE ITEMS TABLE
create table if not exists public.quote_items (
  id uuid primary key default gen_random_uuid(),
  quote_id uuid references public.quotes(id) on delete cascade,
  created_at timestamp with time zone default now(),
  description text not null,
  category text default 'Materials', -- Materials, Labor, Equipment, Disposal
  quantity numeric default 1,
  unit_price numeric not null,
  line_total numeric not null
);

-- 5. ROW LEVEL SECURITY (RLS) POLICIES
alter table public.contractors enable row level security;
alter table public.customers enable row level security;
alter table public.quotes enable row level security;
alter table public.quote_items enable row level security;

-- Public can read a quote only if they have its specific UUID (for client proposal view)
create policy "Public can view individual quotes" on public.quotes
  for select using (true);
